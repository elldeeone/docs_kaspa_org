     1	import { docs } from 'fumadocs-mdx:collections/server';
     2	import { type InferPageType, loader } from 'fumadocs-core/source';
     3	import { lucideIconsPlugin } from 'fumadocs-core/source/lucide-icons';
     4	
     5	// See https://fumadocs.dev/docs/headless/source-api for more info
     6	export const source = loader({
     7	  baseUrl: '/docs',
     8	  source: docs.toFumadocsSource(),
     9	  plugins: [lucideIconsPlugin()],
    10	});
    11	
    12	export function getPageImage(page: InferPageType<typeof source>) {
    13	  const segments = [...page.slugs, 'image.png'];
    14	
    15	  return {
    16	    segments,
    17	    url: `/og/docs/${segments.join('/')}`,
    18	  };
    19	}
    20	
    21	export async function getLLMText(page: InferPageType<typeof source>) {
    22	  const processed = await page.data.getText('processed');
    23	
    24	  return `# ${page.data.title}
    25	
    26	${processed}`;
    27	}
